\newcommand{\VUpapierLargeur}{116mm}

\newcommand{\VUtexteLargeur}{91.69mm}

\newcommand{\VUfolioY}{18.71mm}

% -------------------------------------------------------------------
% 2. TYPE SIZE AND LEADING
% -------------------------------------------------------------------
% Leading of the running text.
%
% THIS VALUE WAS WRONG BY 0.11 pt, AND IT WAS CHECK 11 THAT SHOWED IT.
% The first survey took the MODE OF THE LINE TOPS over 85 pages: 41 px at
% 300 dpi, that is 9.88 pt. But the top of a line depends on the ascenders
% it carries, and the mode of a noisy quantity falls on the nearest
% integer: 41 px was not a measurement, it was a rounding. Measured on the
% BASELINES (tools/baselines.py), which depend on no particular letter,
% the pitch is 41.45 px on leaves 20, 24, 30, 32, 37, 38 -- and 41.26 to
% 41.44 px on the six leaves 62 to 67. 41.45 px = 9.986 pt.
%
% The discrepancy of 0.43 px per line does not show on one line; over the
% forty of a page it makes 17 px, that is 1.4 mm, and that is exactly the
% residue nine pages kept without our knowing where it came from. Set at
% 9.99 pt, it falls away: ten pages failing check 11, five afterwards.
% Neighbouring trials: 9.94 pt leaves eight, 9.96 pt five, 10.02 pt seven.
% We keep the MEASURED value, not the best of the sweep.
\newcommand{\VUinterligne}{9.99pt}

% Size of the running text: the measured x-height is 1.729 mm (4.92 pt
% TeX); the natural width of the surveyed lines is 260.7 pt. Both
% criteria give 10.2 pt for XCharter (cf. docs/journal.md).
\newcommand{\VUcorps}{10.2pt}

% Notes: x-height 1.329 mm, that is 0.766 times the body's, which gives
% 7.81 pt; line pitch 33.25 px = 8.01 pt.
\newcommand{\VUcorpsNote}{7.85pt}
\newcommand{\VUinterligneNote}{8.01pt}

% THE NOTES ARE LETTER-SPACED.
% Two measurements contradicted each other: the x-height gave 7.8 pt for
% the size of the notes, the width of the lines called for 9.3. No size
% satisfies both -- because the original does not play on the size but on
% the tracking: the notes are set letter-spaced, common usage for small
% text of that period.
% Without this, my note lines came out 15 % too short and TeX filled in by
% stretching the word spaces, which gaped.
% Measured on four wholly roman note lines of folio 14: +77, +83, +79, +75
% thousandths of an em. Median kept: +78.
\newcommand{\VUtrackingNote}{78}

% THE WORD SPACE, measured and not inherited from the font.
% Word spaces of the facsimile, over 2011 samples taken from the full
% lines of six pages: median 19.0 px = 4.58 pt, that is 0.449 of the size.
% XCharter gives 2.84 pt of space and 1.42 pt of stretch: its ceiling is
% 4.26 pt, BELOW what the original uses at rest. My lines therefore could
% not fill: TeX left them short (seven "Underfull \hbox") at 94 % of the
% measure, and check 10 reported them without my understanding why.
% It is a post of calibration I had omitted: I had measured the size,
% never the width of the spaces.
%
% BUT 4.58 pt is the value AFTER justification, not at rest: I measured
% the result, not the setting. Set as it stands, it makes the lines that
% did fit overflow (48 failures against 7). We therefore keep a width near
% the font's, with a wide elasticity, which lets each line find its
% measure. That brings the "Underfull \hbox" from seven down to two --
% but does NOT explain the lines at 94 %: the result is insensitive to the
% setting over three values tried. The hypothesis is therefore refuted,
% and the cause remains unknown.
\newcommand{\VUespaceRatio}{0.32}
\newcommand{\VUespaceEtire}{0.30}
\newcommand{\VUespaceSerre}{0.14}

% -------------------------------------------------------------------
% 3. FONT
% -------------------------------------------------------------------
% The facsimile is set in a transitional roman with a large eye, fine
% horizontal serifs, vertical axis.  Among the available fonts, XCharter
% reproduces both the x-height and the width (cf. tools/fonts.tex and
% tools/compare_fonts.tex).
\newcommand{\VUfonte}{XCharter-TLF}

% THE BOLD COMES FROM ANOTHER FAMILY, and that is deliberate.
% The facsimile's bold is a narrow semi-bold, as was usual in the German
% and Luxembourgish fonts of the period: it is not proportioned to its
% roman. XCharter's bold is wide.
% Measured on folio 12, line "la chefa esforco di la voco : ...": the bold
% passage occupies 575 px = 138.5 pt in the facsimile, against 151.4 pt in
% XCharter bold at 10.2 pt, that is 9 % too much -- enough for "neutra."
% no longer to fit in the line.
% Two possible remedies, both tried:
%   - keep XCharter bold and reduce it: one has to come down to 9.2 pt,
%     that is -10 %, which shows (the bold capitals shrink);
%   - take a narrower bold: newtx bold gives 141.5 pt at 10.2 pt, and so
%     comes out right at 10.0 pt, at only -2 %, with a cap height within
%     1.4 % of the measurement.
% It is the second that is kept.
\newcommand{\VUfonteGras}{ntxtlf}
% The bold is defined by a RATIO to the current size, not by an absolute
% size: 10.0 / 10.2 = 0.98. Hard-coded, it set the bold letters of the
% notes (size 7.9 pt) at 10 pt, where they leapt to the eye.
\newcommand{\VUratioGras}{0.98}

\newcommand{\VUauFer}[2]{{\parfillskip=0pt\relax

% THE NOTE CALL OF A TITLE. At folio 5 (leaf 9) the title "Averto" carries

% half a pixel.

\newcommand{\parsuite}{\parfillskip=0pt\relax}

% WORD SPACES ARE UNBREAKABLE.
%
% Without this the model leaks: \nl does force a break at the end of the
% line, but does not stop TeX placing one BEFORE if the matter does not
% fit. A line 0.7 % too wide therefore re-broke itself, in silence,
% without even producing an "Overfull \hbox" -- which happened at folio
% 13. With the spaces made unbreakable, the only lawful breakpoint of a
% line is the one the transcription placed: a line too wide becomes an
% "Overfull \hbox", which check 1 inventories.
% The fault passes from silence to noise.
%
% The spaces keep all their elasticity: the justification is intact.
% The width of the word space, defined OUTSIDE the group where the space
% is made active: written inside, its own spaces became active tokens and
% called it back -- infinite recursion, "TeX capacity exceeded".
\makeatletter\newcommand{\VUespaceMot}{%
  \ifdim\spaceskip=0pt\relax
    \@tempdimb=\f@size pt
    \hskip\VUespaceRatio\@tempdimb
      plus\VUespaceEtire\@tempdimb minus\VUespaceSerre\@tempdimb
  \else \hskip\spaceskip \fi}\makeatother

\endgroup

% 6 bis. VERTICAL JUSTIFICATION OF THE PAGE
% -------------------------------------------------------------------
% The compositor justifies each page VERTICALLY: measured on leaves 20 to
% 39, the last baseline falls 145.3 mm below the folio, to within 0.3 mm,
% whatever the number of lines.
% When the matter does not suffice to fill the page, he distributes the
% surplus as leading. Two means, both visible in the facsimile:
%   -- uniform leads between ALL the lines of the body: the pitch goes
%      from 41.45 px (normal leading, 9.88 pt) to the value proper to the
%      page (leaf 34: 43.10 px = 10.38 pt);
%   -- wider white space at the articulations: end of paragraph, start of
%      an enumeration, resumption of the discourse.
% Without these two settings the page drifts vertically by 40 to 140 px
% (3 to 12 mm): the text stays right line by line, but no line falls
% opposite its model. It is check 11 that measures this.
%
% Leading proper to one page. To be called just after \begin{VUpage},
% BEFORE any matter. Without a call, the page keeps \VUinterligne.
\newcommand{\VUinterlignePage}[1]{%
  \fontsize{\VUcorps}{#1}\selectfont}
% ORDINARY white space between two paragraphs. The facsimile puts one
% everywhere: surveyed on leaves 15 to 44, 27 of the 32 excesses measured
% cluster between 7 and 9 px, median 7.71 px = 1.86 pt. The other values
% are frankly larger (16 px and beyond) and proper to their page.
% It is NOT put in \parskip: at the top of a minipage the glue of
% \parskip survives and would shift the whole page by as much. We
% therefore write it, like the line breaks, where it belongs.
\newcommand{\VUblancAlinea}{\vskip\VUblancAlineaValeur\relax}
\newcommand{\VUblancAlineaValeur}{1.86pt}
% Wider white space at an articulation (opening of an enumeration,
% resumption of the discourse). #1 = the EXCESS measured over the page's
% regular pitch, REDUCED by the ordinary white space above when it
% replaces it.
% Measurement: tools/carding.py.
\newcommand{\VUblanc}[1]{\vskip #1\relax}

% -------------------------------------------------------------------
% 6 ter. TABLES WITH BRACES
% -------------------------------------------------------------------
% The first in the volume is at folio 31 ("GRADI KOMPARALA."). It does not
% let itself be described by an ordinary table: its elements do not form a
% grid. "Superlativo" and "de o ek" fall HALFWAY between two lines, each
% centred on the span of the brace that designates it. The six lines of
% the table, for their part, are properly on the volume's grid: baselines
% at 413, 454, 495 px for the first group, 561, 602.5, 643.5 px for the
% second, that is the current pitch of 41 px.
%
% We therefore keep the volume's rule -- one line of the facsimile makes
% one line of the composed page -- and lay each element at the surveyed
% abscissa.
%
% \VUcase{abscissa}{matter}: matter laid at the given abscissa, counted
% from the left margin, with no width of its own.
\newcommand{\VUcase}[2]{\hbox to 0pt{\hskip#1\relax#2\hss}}
% \VUrang{...}: ONE row of the table, that is one line.
% The whole row is enclosed in a box the width of the block.
% A run of \VUcase left free in the paragraph did not hold: inside
% \VUpage -- where the space is active and the word space elastic -- TeX
% found somewhere to break between the boxes, and each element of the
% table dropped a line, in a staircase. Enclosed in a box, they offer no
% breakpoint at all.
\newcommand{\VUrang}[1]{\hbox to \textwidth{#1\hfil}\penalty-10000\relax}
% \VUtabloCentrita: marks the table that follows as CENTRED on the
% measure. It sets nothing -- in the volume the abscissas are absolute, so
% the centring is already done by the measurement. It exists only for the
% reading page, where a column of arbitrary width has to know what the
% facsimile's measurement states. It is set from the scan, never by eye:
% left margin and right margin equal, and both appreciable (see the
% surveys at folio 220).
\newcommand{\VUtabloCentrita}{}
% \VUdecale{height}{matter}: for what is laid between two lines.
\newcommand{\VUdecale}[2]{\raisebox{#1}[0pt][0pt]{#2}}
% Braces. Those of the facsimile are of the font, not an ornament: they
% are set, they are not extracted from the scan.
% #1 = surveyed height; #2 = vertical displacement of its centre.
% NULL HEIGHT AND DEPTH: laid without that, a brace of 27 pt made a line
% 27 pt tall, and the following row found itself pushed away -- at folio
% 31, the row "egaleso" fell further from "Supereso" than from "infreso".
% The centre of a brace laid on a baseline falls on the mathematical axis,
% that is 10.4 px above that line at 300 dpi: it is from there that the
% surveyed displacements are counted.
\newcommand{\VUaccolade}[2]{%
  \raisebox{#2}[0pt][0pt]{$\left\{\vcenter to #1{}\right.$}}
% CLOSING brace: the one that gathers what is to its left. At folio 31,
% the one for "maxim / minim" designates "de o ek", to its right.
\newcommand{\VUaccoladeD}[2]{%
  \raisebox{#2}[0pt][0pt]{$\left.\vcenter to #1{}\right\}$}}
% HORIZONTAL brace, point at the top: the one from the family tree of
% folio 220, which attaches four names to one. The facsimile gives only
% one in the whole volume. Measurement: ink from x 524 to 1063 (539 px =
% 45.63 mm), y 1339 to 1355, point at the exact middle.
% Like the vertical braces, it is laid with neither height nor depth:
% without that the row of the four names would be pushed away by as much.
% #1 = surveyed width.
\newcommand{\VUaccoladeH}[1]{%
  \raisebox{0pt}[0pt][0pt]{$\overbrace{\hbox to #1{}}$}}
