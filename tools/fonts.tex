% Mesure des metriques des fontes candidates, a 10 pt nominal.
% Sortie : lignes "FONTE <nom> xheight=<pt> capheight=<pt> width_<mot>=<pt>"
\documentclass[10pt]{article}
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
\newlength{\LX}\newlength{\LC}\newlength{\LW}
\newcommand{\mesure}[1]{%
  \settoheight{\LX}{x}%
  \settoheight{\LC}{H}%
  \typeout{FONTE #1 xheight=\the\LX\space capheight=\the\LC}%
  \settowidth{\LW}{La formi Angliano, Franciano, Germaniano, Rusiano,}%
  \typeout{FONTE #1 L1=\the\LW}%
  \settowidth{\LW}{Suediano, Daniano, Poloniano, Grekiano, Arabiano, His-}%
  \typeout{FONTE #1 L2=\the\LW}%
  \settowidth{\LW}{paniano, Skotiano, Laponiano, Suisiano esas tote reguloza,}%
  \typeout{FONTE #1 L3=\the\LW}%
  \settowidth{\LW}{do permisata. Ma per imito dil uzado internaciona, qua por}%
  \typeout{FONTE #1 L4=\the\LW}%
  \settowidth{\LW}{ta nomi di rasi o populi havas formi plu kurta, Ido preferas}%
  \typeout{FONTE #1 L5=\the\LW}%
  \settowidth{\LW}{nacionala dil formo ia (Latina origine) konservas sua formo}%
  \typeout{FONTE #1 L6=\the\LW}%
}
